cmake_minimum_required(VERSION 3.0.2)
project(final_project)
add_compile_options(-std=c++11 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp rospy std_msgs std_srvs geometry_msgs sensor_msgs nav_msgs costmap_2d tf visualization_msgs actionlib actionlib_msgs move_base_msgs control_msgs trajectory_msgs moveit_msgs gazebo_msgs message_generation)
find_package(gazebo REQUIRED)
catkin_python_setup()
add_action_files(FILES Manual.action Delivery.action SemanticPick.action)
add_service_files(FILES SetObjectGrasp.srv)
generate_messages(DEPENDENCIES actionlib_msgs std_msgs)
catkin_package(CATKIN_DEPENDS message_runtime roscpp std_msgs std_srvs actionlib_msgs)
include_directories(${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS} src/third_party/explore_lite/include)
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(final_project_grasp SHARED src/grasp_plugin.cpp)
add_dependencies(final_project_grasp ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(final_project_grasp ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
add_executable(final_project_auto_explore src/automatic_explore.cpp src/third_party/explore_lite/frontier_search.cpp)
target_link_libraries(final_project_auto_explore ${catkin_LIBRARIES})
add_dependencies(final_project_auto_explore ${catkin_EXPORTED_TARGETS})
catkin_install_python(PROGRAMS scripts/mission_server.py scripts/automatic_delivery.py scripts/semantic_mission.py scripts/slam_navigation.py scripts/keyboard_control.py scripts/voice_control.py scripts/velocity_gate.py scripts/yolo_preview.py scripts/cleanup_ros.py tools/render_robot.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS final_project_grasp LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(TARGETS final_project_auto_explore RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES src/third_party/explore_lite/LICENSE.BSD DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/licenses)
install(DIRECTORY launch config urdf worlds maps models voice vision DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
